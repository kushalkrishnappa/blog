**Goal:** Stand up a standalone Docusaurus blog (deployed to GitHub Pages at `https://kushalkrishnappa.github.io/blog/`) that, on every merge to `main`, builds, emits a `posts.json` manifest, deploys, and pings the portfolio's Netlify build hook so the portfolio's preview cards refresh.

## Contract (must match the portfolio consumer exactly)

The portfolio's `src/lib/blog.ts` fetches `https://kushalkrishnappa.github.io/blog/posts.json` and expects:

```json
{
  "generatedAt": "<ISO timestamp>",
  "posts": [
    {
      "slug": "hello-world",
      "title": "Hello, world",
      "date": "2026-07-01",
      "summary": "One-line teaser.",
      "tags": ["distributed-systems"],
      "url": "https://kushalkrishnappa.github.io/blog/hello-world",
      "readingTime": "3 min read"
    }
  ]
}
```

- `posts` sorted newest-first by `date`.
- The portfolio skips any entry lacking string `title` or string `url`; it defaults `slug`→title, `date`/`summary`→"", `tags`→[], `readingTime`→undefined. So those fields should be well-formed here.
- `url` is absolute and MUST equal the post's deployed permalink — guaranteed by requiring an explicit `slug` in frontmatter (Docusaurus uses `${routeBasePath}/${slug}` for the permalink; with `routeBasePath: '/'` and `baseUrl: '/blog/'` the full path is `/blog/${slug}`).

Shared constants used in config and the generator:
- `SITE_URL = "https://kushalkrishnappa.github.io"`
- `BASE_URL = "/blog/"`

```js
// Pure, testable helpers for building posts.json. No filesystem access here.

export function readingTime(text) {
  const words = String(text).trim().split(/\s+/).filter(Boolean).length;
  const minutes = Math.max(1, Math.ceil(words / 200));
  return `${minutes} min read`;
}

function normalizeDate(data, fileName) {
  if (typeof data.date === "string" && data.date) return data.date.slice(0, 10);
  if (data.date instanceof Date) return data.date.toISOString().slice(0, 10);
  const m = typeof fileName === "string" ? fileName.match(/^(\d{4}-\d{2}-\d{2})-/) : null;
  return m ? m[1] : "";
}

function joinUrl(siteUrl, baseUrl, slug) {
  const base = `${siteUrl}${baseUrl}`.replace(/\/+$/, "");
  return `${base}/${String(slug).replace(/^\/+/, "")}`;
}

// data: parsed frontmatter; content: post body; opts: { siteUrl, baseUrl, fileName }
export function toPost(data, content, opts) {
  const { siteUrl, baseUrl, fileName } = opts;
  if (typeof data.slug !== "string" || !data.slug) {
    throw new Error(`Post "${fileName ?? "?"}" is missing a required string \`slug\` in frontmatter`);
  }
  if (typeof data.title !== "string" || !data.title) {
    throw new Error(`Post "${fileName ?? "?"}" is missing a required string \`title\` in frontmatter`);
  }
  return {
    slug: data.slug,
    title: data.title,
    date: normalizeDate(data, fileName),
    summary: typeof data.description === "string" ? data.description : "",
    tags: Array.isArray(data.tags) ? data.tags.filter((t) => typeof t === "string") : [],
    url: joinUrl(siteUrl, baseUrl, data.slug),
    readingTime: readingTime(content),
  };
}

// entries: [{ data, content, fileName }]; opts: { siteUrl, baseUrl }
export function buildManifest(entries, opts) {
  const posts = entries
    .filter((e) => e.data.draft !== true)
    .map((e) => toPost(e.data, e.content, { ...opts, fileName: e.fileName }))
    .sort((a, b) => (a.date < b.date ? 1 : a.date > b.date ? -1 : 0));
  return { generatedAt: new Date().toISOString(), posts };
}
```

- [ ] **Step 6: Implement the IO entrypoint `scripts/generate-manifest.mjs`**

```js
// Reads blog/*.md(x), builds posts.json into the Docusaurus build output.
import { readdirSync, readFileSync, writeFileSync, existsSync } from "node:fs";
import { join } from "node:path";
import matter from "gray-matter";
import { buildManifest } from "./manifest-lib.mjs";

// Keep these in sync with docusaurus.config.ts (url + baseUrl).
const SITE_URL = "https://kushalkrishnappa.github.io";
const BASE_URL = "/blog/";
const BLOG_DIR = "blog";
const OUT_FILE = join("build", "posts.json");

function readEntries(dir) {
  if (!existsSync(dir)) return [];
  return readdirSync(dir, { withFileTypes: true })
    .filter((e) => e.isFile() && /\.mdx?$/.test(e.name))
    .map((e) => {
      const { data, content } = matter(readFileSync(join(dir, e.name), "utf8"));
      return { data, content, fileName: e.name };
    });
}

const manifest = buildManifest(readEntries(BLOG_DIR), { siteUrl: SITE_URL, baseUrl: BASE_URL });
if (!existsSync("build")) {
  throw new Error("build/ not found — run `docusaurus build` before generating the manifest");
}
writeFileSync(OUT_FILE, JSON.stringify(manifest, null, 2));
console.log(`[manifest] wrote ${manifest.posts.length} post(s) to ${OUT_FILE}`);
```

**Files:**

- [ ] **Step 1: Create `.github/workflows/deploy.yml`**

```yaml
name: Deploy blog to GitHub Pages

on:
  push:
    branches: [main]
  workflow_dispatch:

permissions:
  contents: read
  pages: write
  id-token: write

concurrency:
  group: pages
  cancel-in-progress: false

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 22
          cache: npm
      - run: npm ci
      - run: npm run build   # docusaurus build + generate-manifest.mjs (writes build/posts.json)
      - uses: actions/upload-pages-artifact@v3
        with:
          path: build

  deploy:
    needs: build
    runs-on: ubuntu-latest
    environment:
      name: github-pages
      url: ${{ steps.deployment.outputs.page_url }}
    steps:
      - id: deployment
        uses: actions/deploy-pages@v4
      - name: Notify portfolio (Netlify build hook)
        if: ${{ success() }}
        run: |
          if [ -z "${{ secrets.NETLIFY_BUILD_HOOK }}" ]; then
            echo "NETLIFY_BUILD_HOOK not set; skipping portfolio rebuild ping."
          else
            curl -fsS -X POST -d '{}' "${{ secrets.NETLIFY_BUILD_HOOK }}"
            echo "Pinged Netlify build hook."
          fi
```
> Ordering: the `deploy` job runs after `build`, and the Netlify ping is the last step after `actions/deploy-pages`, so Pages (incl. `posts.json`) is live before the portfolio rebuild is triggered.
